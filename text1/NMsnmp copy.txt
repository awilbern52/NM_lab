import json, time
import matplotlib.pyplot as plt
# from pysnmp.hlapi import *
from pysnmp.hlapi import (
    SnmpEngine,
    CommunityData,
    UdpTransportTarget,
    ContextData,
    ObjectType,
    ObjectIdentity,
    nextCmd,
    getCmd
)

COMMUNITY = "public"

ROUTERS = {
    "R1": "10.10.0.1",
    "R2": "10.12.0.12",
    "R3": "10.12.0.13",
    "R4": "10.12.0.21",
    "R5": "10.12.0.1",
}

# OIDs
IF_DESCR = '1.3.6.1.2.1.2.2.1.2'
IF_OPER_STATUS = '1.3.6.1.2.1.2.2.1.8'
IPV4_ADDR = '1.3.6.1.2.1.4.20.1.1'
IPV6_ADDR = '1.3.6.1.2.1.55.1.8.1.2'
CPU_5SEC = '1.3.6.1.4.1.9.2.1.57.0'  # Cisco 5 sec CPU


def snmp_walk(target, oid):
    result = []
    for (errorIndication,
         errorStatus,
         errorIndex,
         varBinds) in nextCmd(
            SnmpEngine(),
            CommunityData(COMMUNITY, mpModel=1),
            UdpTransportTarget((target, 161)),
            ContextData(),
            ObjectType(ObjectIdentity(oid)),
            lexicographicMode=False):

        if errorIndication:
            break
        elif errorStatus:
            break
        else:
            for varBind in varBinds:
                result.append(str(varBind[1]))
    return result


def snmp_get(target, oid):
    iterator = getCmd(
        SnmpEngine(),
        CommunityData(COMMUNITY, mpModel=1),
        UdpTransportTarget((target, 161)),
        ContextData(),
        ObjectType(ObjectIdentity(oid))
    )

    errorIndication, errorStatus, errorIndex, varBinds = next(iterator)

    if errorIndication or errorStatus:
        return None

    return int(varBinds[0][1])


def collect_router_data():
    data = {}

    for name, ip in ROUTERS.items():
        router_data = {}

        interfaces = snmp_walk(ip, IF_DESCR)
        statuses = snmp_walk(ip, IF_OPER_STATUS)
        ipv4 = snmp_walk(ip, IPV4_ADDR)
        ipv6 = snmp_walk(ip, IPV6_ADDR)

        addr_dict = {}
        status_dict = {}

        for i, iface in enumerate(interfaces):
            addr_dict[iface] = {
                "v4": ipv4[i] if i < len(ipv4) else None,
                "v6": ipv6[i] if i < len(ipv6) else None
            }

            if i < len(statuses):
                status_dict[iface] = "up" if statuses[i] == "1" else "down"

        router_data["addresses"] = addr_dict
        router_data["interface_status"] = status_dict

        data[name] = router_data

    return data


def poll_cpu_r1():
    cpu_values = []
    timestamps = []

    for i in range(24):  # 2 minutes / 5 seconds
        cpu = snmp_get(ROUTERS["R1"], CPU_5SEC)
        if cpu is not None:
            cpu_values.append(cpu)
            timestamps.append(i * 5)
        time.sleep(5)

    return timestamps, cpu_values


def plot_cpu(timestamps, cpu_values):
    plt.figure()
    plt.plot(timestamps, cpu_values)
    plt.xlabel("Time (seconds)")
    plt.ylabel("CPU Utilization (%)")
    plt.title("R1 CPU Utilization Over 2 Minutes")
    plt.savefig("R1_CPU.jpg")


def main():
    # collect router data
    router_info = collect_router_data()

    # save json to file
    with open("snmp_output.txt", "w") as f:
        json.dump(router_info, f, indent=4)

    # cpu polling
    timestamps, cpu_values = poll_cpu_r1()
    plot_cpu(timestamps, cpu_values)


if __name__ == "__main__":
    main()
